{%- comment -%} Turns links whose text is their own address into small buttons with an icon and a short label: GitHub repositories show the repository name, doi.org links "DOI", other links the site name. The full address stays in the page as screen-reader text and as the tooltip. Usage: {% include link-pills.html html=... %} {%- endcomment -%} {%- assign parts = include.html | split: '' -%} {%- assign text = after_open | split: "" | first -%} {%- assign tail = after_open | remove_first: text | remove_first: "" -%} {%- if text == url -%} {%- assign host = url | split: "//" | last | split: "/" | first | remove_first: "www." -%} {%- assign path_bits = url | split: "//" | last | split: "/" -%} {%- if host == "github.com" -%} {%- assign icon = "github" -%}{%- assign short = path_bits | last -%} {%- elsif host == "doi.org" -%} {%- assign icon = "paper" -%}{%- assign short = "DOI" -%} {%- else -%} {%- assign icon = "globe" -%}{%- assign short = host -%} {%- endif -%} {% include icon.html name=icon %}{{ short }}{{ text }}{{ tail }} {%- else -%}
